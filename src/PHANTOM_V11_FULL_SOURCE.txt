function clamp(value, min = 0, max = 1) {
  return Math.max(min, Math.min(max, value));
}

function lerp(a, b, amount) {
  return a + (b - a) * amount;
}

function formatTime(seconds) {
  if (!Number.isFinite(seconds) || seconds < 0) return "00:00";
  const total = Math.floor(seconds);
  const hours = Math.floor(total / 3600);
  const minutes = Math.floor((total % 3600) / 60);
  const secs = total % 60;
  if (hours > 0) {
    return `${String(hours).padStart(2, "0")}:${String(minutes).padStart(2, "0")}:${String(secs).padStart(2, "0")}`;
  }
  return `${String(minutes).padStart(2, "0")}:${String(secs).padStart(2, "0")}`;
}

function formatBytes(bytes) {
  if (!Number.isFinite(bytes) || bytes <= 0) return "0 MB";
  const mb = bytes / 1024 / 1024;
  if (mb < 1) return `${Math.max(1, Math.round(bytes / 1024))} KB`;
  return `${mb.toFixed(mb >= 100 ? 0 : 1)} MB`;
}

function getDailyExportCount() {
  try {
    const raw = localStorage.getItem(EXPORT_LIMIT_KEY);
    if (!raw) return { date: new Date().toDateString(), count: 0 };
    const parsed = JSON.parse(raw);
    if (parsed.date !== new Date().toDateString()) {
      return { date: new Date().toDateString(), count: 0 };
    }
    return { date: parsed.date, count: Number(parsed.count) || 0 };
  } catch {
    return { date: new Date().toDateString(), count: 0 };
  }
}

function saveDailyExportCount(count) {
  try {
    localStorage.setItem(
      EXPORT_LIMIT_KEY,
      JSON.stringify({ date: new Date().toDateString(), count })
    );
  } catch {
    // localStorage can be blocked; export still works for the current session.
  }
}

    const glow = ctx.createRadialGradient(
      CANVAS_WIDTH / 2, CANVAS_HEIGHT / 2, 0,
      CANVAS_WIDTH / 2, CANVAS_HEIGHT / 2, Math.min(CANVAS_WIDTH, CANVAS_HEIGHT) * 0.5
    );
    glow.addColorStop(0, `${selectedColors[0]}14`);
    glow.addColorStop(1, "rgba(0,0,0,0)");
    ctx.fillStyle = glow;
    ctx.fillRect(0, 0, CANVAS_WIDTH, CANVAS_HEIGHT);
    ctx.restore();
  };

  const drawGrid = (ctx) => {
    if (!showGrid) return;
    ctx.save();
    ctx.globalAlpha = 0.035;
    ctx.strokeStyle = "#ffffff";
    ctx.lineWidth = 1;
    const step = 100;
    for (let x = 0; x <= CANVAS_WIDTH; x += step) {
      ctx.beginPath();
      ctx.moveTo(x, 0);
      ctx.lineTo(x, CANVAS_HEIGHT);
      ctx.stroke();
    }
    for (let y = 0; y <= CANVAS_HEIGHT; y += step) {
      ctx.beginPath();
      ctx.moveTo(0, y);
      ctx.lineTo(CANVAS_WIDTH, y);
      ctx.stroke();
    }
    ctx.restore();
  };

  const drawCenterImage = (ctx, bass, bassHit, time) => {
    const image = imageRef.current;
    if (!image || !artworkInCenter) return null;

    // The artwork must be rendered relative to the center of the circle.
    // Previously drawContainImage() used a 0,0 origin here, so the image
    // was clipped away and the circle appeared empty.

    const baseRadius = 275 * centerSize;
    const pulse = 1 + bass * 0.045 + bassHit * 0.06;
    const radius = baseRadius * pulse;
    const motion = backgroundMotion ? bassHit * 5 + bass * 1.5 : 0;
    const cx = CANVAS_WIDTH / 2 + Math.sin(time * 0.65) * motion;
    const cy = CANVAS_HEIGHT / 2 + Math.cos(time * 0.56) * motion;

    ctx.save();
    ctx.beginPath();
    ctx.arc(cx, cy, radius, 0, TAU);
    ctx.shadowColor = selectedColors[0];
    ctx.shadowBlur = 34 + bass * 55;
    ctx.strokeStyle = `${selectedColors[0]}80`;
    ctx.lineWidth = 5;
    ctx.stroke();
    ctx.restore();

    ctx.save();
    ctx.beginPath();
    ctx.arc(cx, cy, radius - 2, 0, TAU);
    ctx.clip();

    const innerDiameter = (radius * 2) * 0.91;
    const imageScale = Math.min(
      innerDiameter / image.width,
      innerDiameter / image.height
    );
    const imageWidth = image.width * imageScale;
    const imageHeight = image.height * imageScale;

    ctx.drawImage(
      image,
      cx - imageWidth / 2,
      cy - imageHeight / 2,
      imageWidth,
      imageHeight
    );
    ctx.restore();

    ctx.save();
    ctx.beginPath();
    ctx.arc(cx, cy, radius, 0, TAU);
    ctx.strokeStyle = `${selectedColors[2]}b8`;
    ctx.lineWidth = 2;
    ctx.shadowColor = selectedColors[2];
    ctx.shadowBlur = 18 + bass * 30;
    ctx.stroke();
    ctx.restore();

    return { cx, cy, radius };
  };

  const drawDotRing = (ctx, center, bass, mid, high, bassHit, time) => {
    if (!center) return;
    const data = frequencyDataRef.current;
    const count = Math.max(90, Math.floor(particleCount));
    const baseGap = 24;
    ctx.save();
    for (let i = 0; i < count; i++) {
      const angle = (i / count) * TAU + time * 0.025;
      const index = data ? Math.min(data.length - 1, Math.floor((i / count) * data.length * 0.82)) : 0;
      const frequency = data ? data[index] / 255 : bass;
      const movement = frequency * 58 * visualIntensity + bass * 22 * visualIntensity + bassHit * 90 * visualIntensity;
      const radius = center.radius + baseGap + movement;
      const x = center.cx + Math.cos(angle) * radius;
      const y = center.cy + Math.sin(angle) * radius;
      const size = 2 + frequency * 5.5 + high * 1.8;
      const alpha = 0.22 + frequency * 0.72;
      const color = selectedColors[i % selectedColors.length];
      ctx.globalAlpha = alpha;
      ctx.fillStyle = color;
      ctx.shadowColor = color;
      ctx.shadowBlur = 5 + frequency * 12;
      ctx.beginPath();
      ctx.arc(x, y, size, 0, TAU);
      ctx.fill();
    }
    ctx.restore();
  };

  const drawOrbit = (ctx, center, bass, mid, high, bassHit, time) => {
    if (!center) return;
    ctx.save();
    const rings = 4;
    for (let ring = 0; ring < rings; ring++) {
      const ringBase = center.radius + 46 + ring * 42;
      const count = 42 + ring * 12;
      for (let i = 0; i < count; i++) {
        const angle = (i / count) * TAU + time * (0.06 + ring * 0.025) * (ring % 2 ? -1 : 1);
        const data = frequencyDataRef.current;
        const index = data ? Math.min(data.length - 1, Math.floor((i / count) * data.length * 0.65)) : 0;
        const f = data ? data[index] / 255 : bass;
        const r = ringBase + f * (24 + bassHit * 70) * visualIntensity;
        const x = center.cx + Math.cos(angle) * r;
        const y = center.cy + Math.sin(angle) * r;
        const size = 2 + f * 4 + high * 2;
        const color = selectedColors[(ring + i) % selectedColors.length];
        ctx.globalAlpha = 0.12 + f * 0.65;
        ctx.fillStyle = color;
        ctx.shadowColor = color;
        ctx.shadowBlur = 7;
        ctx.beginPath();
        ctx.arc(x, y, size, 0, TAU);
        ctx.fill();
      }
    }
    ctx.restore();
  };

  const drawPetals = (ctx, center, bass, mid, high, bassHit, time) => {
    if (!center) return;
    ctx.save();
    const petals = 96;
    for (let i = 0; i < petals; i++) {
      const angle = (i / petals) * TAU + time * 0.035;
      const data = frequencyDataRef.current;
      const index = data ? Math.min(data.length - 1, Math.floor((i / petals) * data.length * 0.72)) : 0;
      const f = data ? data[index] / 255 : bass;
      const inner = center.radius + 22;
      const length = 18 + f * 82 * visualIntensity + bassHit * 100 * visualIntensity;
      const outer = inner + length;
      const width = 3 + f * 10;
      const x1 = center.cx + Math.cos(angle) * inner;
      const y1 = center.cy + Math.sin(angle) * inner;
      const x2 = center.cx + Math.cos(angle) * outer;
      const y2 = center.cy + Math.sin(angle) * outer;
      const color = selectedColors[i % selectedColors.length];
      ctx.translate(x1, y1);
      ctx.rotate(angle);
      ctx.globalAlpha = 0.18 + f * 0.6;
      ctx.fillStyle = color;
      ctx.shadowColor = color;
      ctx.shadowBlur = 14;
      ctx.beginPath();
      ctx.moveTo(0, -width);
      ctx.quadraticCurveTo(length * 0.55, -width * 1.8, length, 0);
      ctx.quadraticCurveTo(length * 0.55, width * 1.8, 0, width);
      ctx.closePath();
      ctx.fill();
      ctx.setTransform(1, 0, 0, 1, 0, 0);
    }
    ctx.restore();
  };

  const drawShards = (ctx, center, bass, mid, high, bassHit, time) => {
    if (!center) return;
    const count = 110;
    const data = frequencyDataRef.current;
    ctx.save();
    for (let i = 0; i < count; i++) {
      const angle = (i / count) * TAU + time * 0.018;
      const index = data ? Math.min(data.length - 1, Math.floor((i / count) * data.length * 0.7)) : 0;
      const f = data ? data[index] / 255 : bass;
      const inner = center.radius + 30;
      const length = 15 + f * 120 * visualIntensity + bassHit * 95 * visualIntensity;
      const outer = inner + length;
      const width = 2 + f * 7;
      const x1 = center.cx + Math.cos(angle) * inner;
      const y1 = center.cy + Math.sin(angle) * inner;
      const x2 = center.cx + Math.cos(angle) * outer;
      const y2 = center.cy + Math.sin(angle) * outer;
      const color = selectedColors[i % selectedColors.length];
      ctx.globalAlpha = 0.12 + f * 0.72;
      ctx.strokeStyle = color;
      ctx.lineWidth = width;
      ctx.shadowColor = color;
      ctx.shadowBlur = 10;
      ctx.beginPath();
      ctx.moveTo(x1, y1);
      ctx.lineTo(x2, y2);
      ctx.stroke();
    }
    ctx.restore();
  };

  const drawHalo = (ctx, center, bass, mid, high, bassHit, time) => {
    if (!center) return;
    ctx.save();
    const rings = 5;
    for (let i = 0; i < rings; i++) {
      const radius = center.radius + 28 + i * 34 + bass * 22 + bassHit * 55;
      const alpha = 0.07 + (1 - i / rings) * 0.12 + mid * 0.12;
      ctx.globalAlpha = alpha;
      ctx.strokeStyle = selectedColors[i % selectedColors.length];
      ctx.lineWidth = 2 + high * 2;
      ctx.shadowColor = selectedColors[i % selectedColors.length];
      ctx.shadowBlur = 22 + bass * 20;
      ctx.beginPath();
      ctx.arc(center.cx, center.cy, radius + Math.sin(time * 1.3 + i) * 5, 0, TAU);
      ctx.stroke();
    }
    ctx.restore();
  };

  const drawParticles = (ctx, center, high, bassHit, time) => {
    if (!center) return;
    ctx.save();
    const count = Math.floor(60 + particleCount * 0.28);
    for (let i = 0; i < count; i++) {
      const angle = (i * 137.5 * Math.PI) / 180;
      const distance = center.radius + 150 + ((i * 67) % 520);
      const pulse = Math.sin(time * 0.9 + i * 0.7) * 7;
      const x = center.cx + Math.cos(angle) * (distance + pulse);
      const y = center.cy + Math.sin(angle) * (distance + pulse);
      const size = 1.2 + high * 2.6;
      ctx.globalAlpha = 0.025 + high * 0.2 + bassHit * 0.18;
      ctx.fillStyle = selectedColors[i % selectedColors.length];
      ctx.beginPath();
      ctx.arc(x, y, size, 0, TAU);
      ctx.fill();
    }
    ctx.restore();
  };

  const drawWatermark = (ctx) => {
    if (!showWatermark) return;
    ctx.save();
    const size = Math.max(18, Math.round(Math.min(CANVAS_WIDTH, CANVAS_HEIGHT) * 0.015));
    ctx.globalAlpha = 0.72;
    ctx.fillStyle = "#ffffff";
    ctx.font = `800 ${size}px Inter, Arial, sans-serif`;
    ctx.textAlign = "center";
    ctx.shadowColor = selectedColors[0];
    ctx.shadowBlur = 18;
    ctx.fillText("PHANTOM", CANVAS_WIDTH / 2, CANVAS_HEIGHT - Math.round(size * 1.9));
    ctx.globalAlpha = 0.16;
    ctx.font = `500 ${Math.max(8, Math.round(size * 0.38))}px Inter, Arial, sans-serif`;
    ctx.letterSpacing = "5px";
    ctx.shadowBlur = 0;
    ctx.fillText("AUDIO VISUAL ENGINE", CANVAS_WIDTH / 2, CANVAS_HEIGHT - Math.round(size * 0.8));
    ctx.restore();
  };

  const drawFrame = useCallback(
    (timestamp) => {
      const canvas = canvasRef.current;
      if (!canvas) return;
      const ctx = canvas.getContext("2d", { alpha: false });
      const { bass, mid, high, bassHit } = getAudioValues();
      const time = timestamp * 0.001;

      ctx.clearRect(0, 0, CANVAS_WIDTH, CANVAS_HEIGHT);
      drawBackground(ctx, bass, bassHit, time);
      drawGrid(ctx);
      const center = drawCenterImage(ctx, bass, bassHit, time);

      if (visualStyle === "dots") drawDotRing(ctx, center, bass, mid, high, bassHit, time);
      if (visualStyle === "orbit") drawOrbit(ctx, center, bass, mid, high, bassHit, time);
      if (visualStyle === "petals") drawPetals(ctx, center, bass, mid, high, bassHit, time);
      if (visualStyle === "shards") drawShards(ctx, center, bass, mid, high, bassHit, time);
      if (visualStyle === "halo") drawHalo(ctx, center, bass, mid, high, bassHit, time);
      if (visualStyle === "hybrid") {
        drawDotRing(ctx, center, bass, mid, high, bassHit, time);
        drawHalo(ctx, center, bass, mid, high, bassHit, time);
      }

      drawParticles(ctx, center, high, bassHit, time);
      drawWatermark(ctx);

      if (captureTrackRef.current?.requestFrame && exporting) {
        const now = performance.now();
        const interval = 1000 / Math.max(1, exportFps);
        if (now - captureLastFrameRef.current >= interval - 1) {
          captureLastFrameRef.current = now;
          captureTrackRef.current.requestFrame();
        }
      }

      animationRef.current = requestAnimationFrame(drawFrame);
    },
    [visualStyle, colorPreset, backgroundMotion, backgroundBlur, visualIntensity, particleCount, centerSize, showGrid, showWatermark, selectedColors, exporting, exportFps, useArtworkAsBackground, artworkPlacement]
  );

  useEffect(() => {

      animationRef.current = null;

    animationRef.current = requestAnimationFrame(drawFrame);
    return () => {
      if (animationRef.current) cancelAnimationFrame(animationRef.current);
      animationRef.current = null;
    };
  }, [screen, drawFrame]);

  const exportVideo = async () => {
    if (exporting) return;

    const canvas = canvasRef.current;

    if (!artworkReady) {

      return;

    const config = getExportConfig();
    const mimeType = createRecorderMime();
    if (!mimeType) {
      alert("This browser cannot export WebM video. Try Chrome or Edge.");
      return;
    }

    const oldWidth = CANVAS_WIDTH;
    const oldHeight = CANVAS_HEIGHT;
    setExporting(true);
    setExportError("");
    setExportProgress(0);

    try {
      await setupAudio();

      const videoTracks = canvasStream.getVideoTracks();
      const captureTrack = videoTracks[0] || null;
      captureTrackRef.current = captureTrack;

      const audioTracks = destination.stream.getAudioTracks();
      const stream = new MediaStream([...videoTracks, ...audioTracks]);
      const recorder = new MediaRecorder(stream, {
        mimeType,
        videoBitsPerSecond: config.bitrate,
        audioBitsPerSecond: 192_000,
      });

      const chunks = [];
      let stopped = false;
      let progressFrame = null;
      let exportTimer = null;
      let lastRendered = performance.now();
      const frameInterval = 1000 / Math.max(1, config.fps);

      const cleanup = () => {
        videoTracks.forEach((track) => track.stop());
        audio.removeEventListener("ended", onEnded);
        captureTrackRef.current = null;
        if (progressFrame) cancelAnimationFrame(progressFrame);
        if (exportTimer) clearTimeout(exportTimer);
      };

      const restoreCanvas = () => {
        CANVAS_WIDTH = oldWidth;
        CANVAS_HEIGHT = oldHeight;
        canvas.width = oldWidth;
        canvas.height = oldHeight;
      };

      const finishExport = () => {
        if (stopped) return;
        stopped = true;
        if (recorder.state !== "inactive") recorder.stop();
      };

      const onEnded = () => finishExport();

      const renderExportFrame = () => {
        if (stopped) return;
        const now = performance.now();
        if (now - lastRendered >= frameInterval - 1) {
          lastRendered = now;
          drawFrame(now);
          if (captureTrack?.requestFrame) captureTrack.requestFrame();
        }
        exportTimer = setTimeout(renderExportFrame, Math.max(1, frameInterval * 0.55));
      };

      recorder.ondataavailable = (event) => {
        if (event.data?.size) chunks.push(event.data);
      };

      recorder.onerror = (event) => {
        console.error("PHANTOM export error", event);
        cleanup();
        restoreCanvas();
        setExporting(false);
        setExportError("Export failed. Try a shorter track or another browser.");
      };

      recorder.onstop = () => {
        cleanup();
        restoreCanvas();
        const blob = new Blob(chunks, { type: mimeType });
        const url = URL.createObjectURL(blob);

        if (resultVideoUrlRef.current) URL.revokeObjectURL(resultVideoUrlRef.current);
        resultVideoUrlRef.current = url;
        setExportedBlob(blob);
        setExportedUrl(url);
        setExportedSize(blob.size);
        setExportedName(`phantom-${config.file}-${Date.now()}-${config.width}x${config.height}-${config.fps}fps.webm`);
        setExportProgress(100);
        setExporting(false);
        setIsPlaying(false);
        setScreen("result");

        const nextCount = exportCount + 1;
        setExportCount(nextCount);
        saveDailyExportCount(nextCount);
      };

      audio.addEventListener("ended", onEnded);
      recorder.start(500);
      setIsPlaying(true);
      await audio.play();
      await playBackgroundVideo();

      // Dedicated export render loop: the frames are generated from the same audio clock
      // that drives the analyser, and the capture track is explicitly requested at the chosen FPS.
      lastRendered = performance.now() - frameInterval;
      renderExportFrame();

      const updateProgress = () => {
        if (audio.duration && !audio.paused) {
          setExportProgress(Math.min(99, Math.floor((audio.currentTime / audio.duration) * 100)));
          progressFrame = requestAnimationFrame(updateProgress);
        }
      };
      updateProgress();
    } catch (error) {
      console.error("PHANTOM EXPORT ERROR:", error);
      captureTrackRef.current = null;
      CANVAS_WIDTH = oldWidth;
      CANVAS_HEIGHT = oldHeight;
      if (canvas) {
        canvas.width = oldWidth;
        canvas.height = oldHeight;
      }
      setExporting(false);
      setIsPlaying(false);
      setExportError("Could not export the video. Check browser support and try again.");
    }
  };
